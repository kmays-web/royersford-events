const { useState, useEffect, useMemo } = React;


// EDIT THESE LOCAL EVENTS ANYTIME - Add your featured Royersford events here

function EventFinder() {

  const [search, setSearch] = useState("");

  const [category, setCategory] = useState("All");

  const [allEvents, setAllEvents] = useState([...localEvents]);

  const [loading, setLoading] = useState(false);

  const [hasSearched, setHasSearched] = useState(false);


  const categories = ["All", "Concerts", "Sports", "Festivals", "Arts", "Food", "Community", "Family"];


  // Search Ticketmaster + Eventbrite

  const searchEvents = async () => {

    setLoading(true);

    setHasSearched(true);

    

    try {

      const apiEvents = [];

      

      // Ticketmaster API - FREE (5000 calls/day)

      const tmResponse = await fetch(`https://app.ticketmaster.com/discovery/v2/events.json?apikey=Vo2OY2CCcxx6WwOMa9a0m8thGnGMFIr5&city=Royersford&stateCode=PA&radius=25&size=50`);

      const tmData = await tmResponse.json();

      

      if (tmData._embedded?.events) {

        tmData._embedded.events.forEach(event => {

          apiEvents.push({

            id: `tm-${event.id}`,

            name: event.name,

            date: event.dates.start.localDate,

            city: event._embedded?.venues?.[0]?.city?.name || "Royersford",

            state: event._embedded?.venues?.[0]?.state?.stateCode || "PA",

            category: event.classifications?.[0]?.segment?.name || "Entertainment",

            description: event.info || event.pleaseNote || `${event.name} at ${event._embedded?.venues?.[0]?.name || 'venue'}`,

            price: event.priceRanges ? `${event.priceRanges[0].min}-${event.priceRanges[0].max}` : "See Website",

            venue: event._embedded?.venues?.[0]?.name || "",

            url: event.url,

            source: "Ticketmaster"

          });

        });

      }

      

      // Eventbrite API - FREE (1000 calls/hour)

      try {

        const ebResponse = await fetch(`https://www.eventbriteapi.com/v3/events/search/?location.address=Royersford,PA&location.within=25mi&token=T6RTGNB5BNIHV7SUJFME`);

        const ebData = await ebResponse.json();

        

        if (ebData.events) {

          ebData.events.forEach(event => {

            apiEvents.push({

              id: `eb-${event.id}`,

              name: event.name.text,

              date: event.start.local.split('T')[0],

              city: event.venue?.address?.city || "Royersford",

              state: event.venue?.address?.region || "PA",

              category: event.category?.name || "Community",

              description: event.description?.text?.substring(0, 150) || event.summary || "See event page for details",

              price: event.is_free ? "Free" : "See Website",

              venue: event.venue?.name || "",

              url: event.url,

              source: "Eventbrite"

            });

          });

        }

      } catch (error) {

        console.log("Eventbrite search completed");

      }

      

      // Combine with local events

      setAllEvents([...localEvents, ...apiEvents]);

      

    } catch (error) {

      console.error("Error fetching events:", error);

    } finally {

      setLoading(false);

    }

  };


  const filtered = useMemo(() => {

    return allEvents.filter(e => {

      const matchSearch = !search || 

        e.name.toLowerCase().includes(search.toLowerCase()) || 

        e.description.toLowerCase().includes(search.toLowerCase()) ||

        e.venue?.toLowerCase().includes(search.toLowerCase());

      const matchCat = category === "All" || e.category === category;

      return matchSearch && matchCat;

    }).sort((a, b) => new Date(a.date) - new Date(b.date));

  }, [allEvents, search, category]);


  return (

    <div className="bg-gradient-to-br from-purple-50 via-blue-50 to-indigo-100 py-8 min-h-screen">

      <div className="max-w-7xl mx-auto px-4">

        

        {/* Search Box */}

        <div className="bg-white rounded-2xl shadow-xl p-6 mb-8">

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-4">

            

            <input 

              type="text" 

              placeholder="Search events..." 

              value={search} 

              onChange={e => setSearch(e.target.value)}

              className="w-full px-4 py-3 border-2 border-indigo-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-indigo-500"

            />

            

            <select 

              value={category} 

              onChange={e => setCategory(e.target.value)}

              className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-indigo-500 bg-white"

            >

              {categories.map(c => <option key={c} value={c}>{c}</option>)}

            </select>

            

            <button 

              onClick={searchEvents}

              disabled={loading}

              className="bg-gradient-to-r from-indigo-600 to-purple-600 hover:from-indigo-700 hover:to-purple-700 text-white py-3 px-6 rounded-lg font-semibold transition-all disabled:opacity-50"

            >

              {loading ? "Searching..." : hasSearched ? "Refresh Events" : "Search Events"}

            </button>

          </div>

          

          <div className="flex justify-between items-center text-sm">

            <p className="text-gray-500">

              🎯 Searching Ticketmaster, Eventbrite & local Royersford venues

            </p>

            <p className="text-gray-600">

              <span className="font-semibold text-indigo-600">{filtered.length}</span> events found

            </p>

          </div>

        </div>


        {/* Loading */}

        {loading && (

          <div className="text-center py-16">

            <div className="w-12 h-12 mx-auto mb-4 border-4 border-indigo-200 border-t-indigo-600 rounded-full animate-spin"></div>

            <p className="text-gray-600 text-lg">Searching for events...</p>

          </div>

        )}


        {/* Events Grid */}

        {!loading && (

          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">

            {filtered.map(e => (

              <div key={e.id} className="bg-white rounded-xl shadow-md hover:shadow-2xl transition-all duration-300 border border-gray-100">

                <div className="p-6">

                  

                  {/* Header */}

                  <div className="flex justify-between items-start mb-3">

                    <h3 className="text-xl font-bold text-gray-800 flex-1 pr-2">{e.name}</h3>

                    <span className="bg-gradient-to-r from-indigo-100 to-purple-100 text-indigo-700 text-xs px-3 py-1 rounded-full font-semibold whitespace-nowrap">

                      {e.category}

                    </span>

                  </div>

                  

                  {/* Source Badge */}

                  {e.source && (

                    <div className="mb-2">

                      <span className="bg-green-100 text-green-700 text-xs px-2 py-1 rounded font-semibold">

                        ⭐ {e.source}

                      </span>

                    </div>

                  )}

                  

                  {/* Description */}

                  <p className="text-gray-600 text-sm mb-4 line-clamp-3">{e.description}</p>

                  

                  {/* Details */}

                  <div className="space-y-2 mb-4">

                    <div className="text-gray-700 text-sm flex items-start">

                      <span className="mr-2">📅</span>

                      <span>

                        {new Date(e.date).getFullYear() === 2025 && new Date(e.date).getMonth() === 11 ? 

                          "Year-Round / Check Website" : 

                          new Date(e.date).toLocaleDateString('en-US', {month:'long', day:'numeric', year:'numeric'})

                        }

                      </span>

                    </div>

                    <div className="text-gray-700 text-sm flex items-start">

                      <span className="mr-2">📍</span>

                      <span>{e.venue || `${e.city}, ${e.state}`}</span>

                    </div>

                    <div className="text-gray-700 text-sm font-semibold flex items-start">

                      <span className="mr-2">💵</span>

                      <span>{e.price}</span>

                    </div>

                  </div>

                  

                  {/* CTA Button */}

                  {e.url ? (

                    <a 

                      href={e.url} 

                      target="_blank" 

                      rel="noopener noreferrer"

                      className="block w-full bg-gradient-to-r from-indigo-600 to-purple-600 hover:from-indigo-700 hover:to-purple-700 text-white py-2 px-4 rounded-lg text-center font-semibold transition-all"

                    >

                      View Details →

                    </a>

                  ) : (

                    <div className="w-full bg-gray-100 text-gray-500 py-2 px-4 rounded-lg font-semibold text-center">

                      More Info Coming Soon

                    </div>

                  )}

                </div>

              </div>

            ))}

          </div>

        )}


        {/* No Results */}

        {!loading && filtered.length === 0 && (

          <div className="text-center py-16 bg-white rounded-2xl shadow-lg">

            <p className="text-gray-500 text-xl mb-2">No events found</p>

            <p className="text-gray-400">Try adjusting your search or category filter</p>

          </div>

        )}


        {/* Submit Event CTA */}

        <div className="mt-12 bg-gradient-to-r from-indigo-50 to-purple-50 border-2 border-indigo-200 rounded-2xl p-8 text-center">

          <h3 className="text-2xl font-bold text-indigo-900 mb-3">

            📢 Have an Event to Add?

          </h3>

          <p className="text-indigo-700 mb-4 max-w-2xl mx-auto">

            Are you hosting an event in Royersford, Limerick, or Upper Providence? 

            We'd love to feature it! Local businesses and organizations - contact us to get listed.

          </p>

          <div className="flex flex-wrap justify-center gap-4">

            <a 

              href="mailto:karilen@karilenmays.com?subject=Add%20Event%20to%20Royersford%20Events" 

              className="inline-block bg-indigo-600 hover:bg-indigo-700 text-white py-3 px-8 rounded-lg font-bold text-lg transition-all shadow-lg hover:shadow-xl"

            >

              Submit Your Event

            </a>

            <a 

              href="mailto:karilen@karilenmays.com?subject=Partnership%20Inquiry" 

              className="inline-block bg-white hover:bg-gray-50 text-indigo-600 border-2 border-indigo-600 py-3 px-8 rounded-lg font-bold text-lg transition-all"

            >

              Partner With Us

            </a>

          </div>

        </div>


        {/* Footer Note */}

        <div className="mt-8 text-center text-sm text-gray-500">

          <p>Events powered by Ticketmaster, Eventbrite, and local Royersford partners</p>

          <p className="mt-1">Updated regularly • Free to use • Community-focused</p>

        </div>

      </div>

    </div>

  );

}
